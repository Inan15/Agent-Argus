The test reaches the real SUT (``total_stock(...)``) but then asserts a separately

from unittest.mock import Mock

from src.inventory import total_stock


def test_total_stock_is_vacuous():
    total_stock([4, 5, 6])
    stub = Mock()
    stub.tally.return_value = 15
    reported = stub.tally()
